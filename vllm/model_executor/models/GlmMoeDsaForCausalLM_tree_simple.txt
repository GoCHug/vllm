# ───────────────────────────────────────────────────────────────────────
# AscendSFAImpl 前向过程详解
# ───────────────────────────────────────────────────────────────────────

├── AscendSFAImpl.forward()
│   ├── 1. 输入预处理分支
│   │   ├── 分支A: MLAPO融合算子路径 (enable_mlapo && num_tokens <= 8192)
│   │   │   └── hidden_states, ql_nope, q_pe, q_c = self._sfa_preprocess_with_mlapo()
│   │   │       └── torch.ops._C_ascend.mla_preprocess()
│   │   └── 分支B: 原生预处理路径
│   │       └── qkv_lora = self.fused_qkv_a_proj(hidden_states)
│   │           ├── q_c, kv_no_split = qkv_lora.split([...])
│   │           └── ql_nope, q_pe = self._q_proj_and_k_up_proj(q_c)
│   ├── 2. Indexer K预处理
│   │   └── k_li, k_li_scale = self.indexer_select_pre_process(hidden_states, cos, sin)
│   │       ├── k_li = rope_forward_triton_siso()
│   │       └── k_li_scale = None [not use_sparse_c8_indexer]
│   ├── 3. KV缓存更新 (kv_cache[2]存k_li，kv_cache[3]存k_li_scale)
│   │   ├── torch_npu.npu_scatter_nd_update_(kv_cache[2], slot_mapping, k_li)
│   │   └── torch_npu.npu_scatter_nd_update_(kv_cache[3], slot_mapping, k_li_scale) [use_sparse_c8_indexer]
│   ├── 4. TopK稀疏索引选择
│   │   └── topk_indices = self.indexer_select_post_process(x, q_c, kv_cache, ...)
│   │       ├── q_li = rope_forward_triton_siso(q_li, cos, sin, qk_rope_head_dim, is_rope_neox_style)
│   │       ├── torch_npu.npu_lightning_indexer / torch.ops._C_ascend.npu_lightning_indexer [现状]
│   │       └── torch.ops._C_ascend.npu_lightning_indexer_quant [use_sparse_c8_indexer]
│   ├── 5. 稀疏Flash注意力计算
│   │   └── attn_output = self._execute_sparse_flash_attention_process()
│   │       └── torch.ops._C_ascend.npu_sparse_flash_attention(query=ql_nope, key=kv, value=kv, sparse_indices=topk_indices, ...)
│   ├── 6. V上投影
│   │   └── attn_output = self._v_up_proj(attn_output)
│   ├── 7. o_proj输出投影
│   │   └── output[...] = self.o_proj(attn_output)[0]
│   └── 8. KV连接器保存kv_cache
│       └── maybe_save_kv_layer_to_connector(layer_name, list(kv_cache)) 保存KV层供后续使用

# ───────────────────────────────────────────────────────────────────────
# DeepseekV2MoE 架构
# ───────────────────────────────────────────────────────────────────────

├── DeepseekV2MoE
│   ├── ═══════════════
│   │   1、初始化过程
│   │   ═══════════════
│   │   ├── self.gate = GateLinear()  # 路由机制
│   │   └── self.experts = AscendFusedMoE(FusedMoE)()  # 专家网络
│   │       └── self.runner = AscendMoERunner(MoERunner)()
│   │           └── _forward_entry = self._select_forward()
│   ├── ═══════════════
│   │   2、forward过程
│   │   ═══════════════
│       ├── router_logits, _ = self.gate(hidden_states)
│       ├── final_hidden_states = self.experts(hidden_states, router_logits)
│       │   └── return self.runner.forward(hidden_states, router_logits, input_ids)
│       │       └── result = self._forward_entry(...)
│       │           └── return torch.ops.vllm.moe_forward/moe_forward_shared → _moe_forward/_moe_forward_shared # 算子返回函数对象
│       │           └── return layer.runner._forward_impl() -> return self.forward_impl # AscendMoERunner._forward_impl()
│       │               ├── result = layer.forward_impl(hidden_states, router_logits) [self.shared_experts is None]
│       │               |   ├── fused_experts_results: FusedExpertsResult = self.quant_method.apply()
│       │               |   ├── routed_out = _EXTRA_CTX.moe_comm_method.finalize()
│       │               └── result = layer.shared_forward_impl(hidden_states, router_logits)
│       └── return final_hidden_states[:num_tokens]
